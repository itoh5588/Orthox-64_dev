# Orthox-64 aarch64 を Raspberry Pi 4 で起動する最小の設定。
#
# **SD カードの先頭パーティション (FAT32 / boot) の直下に置く。**
# Raspberry Pi OS のカードなら、そこにある config.txt を退避して差し替える。

# 64bit で起動する。**これが無いと 32bit で始まり、いきなり迷子になる**
arm_64bit=1

# 読み込むカーネル。既定は kernel8.img だが明示しておく
kernel=kernel8.img

# ---- シリアルコンソール ----------------------------------------------------
#
# **Pi 4 の PL011 (UART0) は既定で Bluetooth に取られている。**
# GPIO14/15 に出ているのは mini-UART のほう。
# disable-bt で Bluetooth を外し、**PL011 を GPIO14/15 に回す**。
# これでカーネル側は GPIO の alt-func もクロックも触らなくてよい
# (ファームウェアがやる)。
enable_uart=1
dtoverlay=disable-bt

# ファームウェア自身のログもシリアルに出す。**最初の 1 回は必ず入れること** —
# カーネルから 1 文字も出ないとき、「配線が悪い」のか「カーネルが動いて
# いない」のかを切り分けられる
uart_2ndstage=1

# ---- 起動を単純にする ------------------------------------------------------

# 追加のコアを止めておく。**カーネルは CPU 1 本前提**なので、
# 4 コアが同じ入口に飛んでくると壊れる。
# (start.S は MPIDR を見て 0 番以外を止める作りだが、まずは外側でも絞る)
#
# ※ Pi 4 のファームウェアはこの指定を無視することがある。start.S 側の
#   判定が本命で、これは念のため
enable_gic=1

# 起動を速くする。実機で待たされると切り分けが遅くなる
boot_delay=0
disable_splash=1
